CREATE TABLE roles (
    id INT IDENTITY(1,1) PRIMARY KEY,
    name NVARCHAR(50) NOT NULL UNIQUE,
    description NVARCHAR(255),
    is_active BIT DEFAULT 1
);
-- Index on is_active for filtering roles
CREATE INDEX idx_roles_is_active ON roles(is_active);


CREATE TABLE waste_types (
    id INT IDENTITY(1,1) PRIMARY KEY,
    name NVARCHAR(50) NOT NULL UNIQUE,
    description NVARCHAR(255),
    color_code NVARCHAR(10) DEFAULT '#FFFFFF',
    icon_path NVARCHAR(255)
);
-- Index on color_code if querying by color is common
CREATE INDEX idx_waste_types_color_code ON waste_types(color_code);


CREATE TABLE users (
    id INT IDENTITY(1,1) PRIMARY KEY,
    username NVARCHAR(50) NOT NULL UNIQUE,
    password_hash NVARCHAR(255) NOT NULL,
    role_id INT NULL,
    email NVARCHAR(100) UNIQUE,
    is_active BIT DEFAULT 1,
    last_login DATETIME,
    created_at DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_users_roles FOREIGN KEY (role_id) REFERENCES roles(id)
);
-- Indexes on foreign key and commonly filtered columns
CREATE INDEX idx_users_role_id ON users(role_id);
CREATE INDEX idx_users_is_active ON users(is_active);
CREATE INDEX idx_users_last_login ON users(last_login);


CREATE TABLE images (
    id INT IDENTITY(1,1) PRIMARY KEY,
    image_path NVARCHAR(255) NOT NULL,
    original_name NVARCHAR(255),
    file_size INT,
    mime_type NVARCHAR(50),
    is_temporary BIT DEFAULT 0,
    uploaded_at DATETIME DEFAULT GETDATE(),
    uploaded_by INT NULL,
    CONSTRAINT FK_images_users FOREIGN KEY (uploaded_by) REFERENCES users(id) ON DELETE SET NULL
);
-- Indexes on foreign key and filtering columns
CREATE INDEX idx_images_uploaded_by ON images(uploaded_by);
CREATE INDEX idx_images_is_temporary ON images(is_temporary);
CREATE INDEX idx_images_uploaded_at ON images(uploaded_at);


CREATE TABLE classification_results (
    id INT IDENTITY(1,1) PRIMARY KEY,
    image_id INT NOT NULL,
    waste_type_id INT NOT NULL,
    classified_at DATETIME DEFAULT GETDATE(),
    confidence_score FLOAT,
    model_version NVARCHAR(50) DEFAULT 'v1.0',
    processing_time_ms INT DEFAULT 0,
    status NVARCHAR(20) DEFAULT 'success',
    CONSTRAINT FK_classification_images FOREIGN KEY (image_id) REFERENCES images(id) ON DELETE CASCADE,
    CONSTRAINT FK_classification_waste_types FOREIGN KEY (waste_type_id) REFERENCES waste_types(id)
);
-- Indexes on foreign keys and status for filtering
CREATE INDEX idx_classification_image_id ON classification_results(image_id);
CREATE INDEX idx_classification_waste_type_id ON classification_results(waste_type_id);
CREATE INDEX idx_classification_status ON classification_results(status);


CREATE TABLE feedbacks (
    id INT IDENTITY(1,1) PRIMARY KEY,
    classification_id INT NOT NULL,
    user_id INT NULL,
    feedback_text NVARCHAR(255),
    feedback_type NVARCHAR(20) DEFAULT 'neutral',
    created_at DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_feedbacks_classification FOREIGN KEY (classification_id) REFERENCES classification_results(id),
    CONSTRAINT FK_feedbacks_users FOREIGN KEY (user_id) REFERENCES users(id)
);
-- Indexes on foreign keys and feedback type
CREATE INDEX idx_feedbacks_classification_id ON feedbacks(classification_id);
CREATE INDEX idx_feedbacks_user_id ON feedbacks(user_id);
CREATE INDEX idx_feedbacks_feedback_type ON feedbacks(feedback_type);
